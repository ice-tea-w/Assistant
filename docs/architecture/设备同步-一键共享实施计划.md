# 设备同步「一键共享 / 一键连接」实施计划

> **For agentic workers:** 按任务顺序执行，每个任务结束跑一次测试并提交。步骤用 `- [ ]` 勾选跟踪。

**Goal:** 让「设备同步」页在主机角色下点一下就开好 Windows 共享、在连接方角色下点一下就连上，不再要求用户手敲 PowerShell。

**Architecture:** 不写任何网络代码，把 Windows 自带的 `net share` / `net use` 当 API 调用。共享名与命令行拼装抽成纯函数（`SyncCommands`）便于单测；提权调用抽成 `Os::runElevated`（`ShellExecuteExW` + `runas` + 等退出码）；`SyncBridge` 只做编排，UI 只做展示。`FolderTransport::probe()` 的失败原因显式化（`probeError()`），否则按钮点了也不知道为什么失败。

**Tech Stack:** C++20 / Qt6 (Quick, Test) / Win32 (`shellapi.h`) / CMake + vcpkg。

**现状（动手前必须知道）：**
- 同步模块零 socket（只有 `src/ui/SyncBridge.cpp:191-195` 的 `QNetworkInterface` 读本机 IP）。不要引入网络代码。
- 页面现有 `SyncPage.qml:436` 的 `net share` 命令把路径硬编码成 `E:\SyncHub`，且只用 `/grant:everyone,full`（不含 NTFS）—— 本计划要修掉。
- `SyncBridge::checkShared()`（`src/ui/SyncBridge.cpp:675-721`）失败时只返回一句「共享目录不可访问或不可写」，`FolderTransport::probe()`（`src/core/sync/FolderTransport.cpp:72-90`）只返回 `bool`。
- 构建：`CMakeLists.txt` 用 `file(GLOB_RECURSE)`，新增 `.cpp` 必须重新 configure。
- 测试：QTest，`.\build-main\Debug\assistant_tests.exe -o result.txt,txt`（终端直接输出中文会乱码）。

**已定的决策（需用户点头）：**
1. **提权方式**：`net share` 需要管理员。用 `ShellExecuteExW(lpVerb=L"runas")` 触发 UAC，用户点「是」才继续；取消则返回 -1000 并提示，不重试、不换招。
2. **连接方不写密码文件**：`net use` 用 `QProcess::start("net", args)`（参数列表，无 shell 拼接），密码只在进程参数里出现（进程列表可见，UI 上注明）。密码可选，留空即用当前登录凭据。
3. **不改变 `probe()` 语义**：`probeError()` 返回空串时 `probe()` 为 `true`，老调用点不受影响。

`src/core/sync/SyncCommands.cpp`：

```cpp
#include "core/sync/SyncCommands.h"

#include <QDir>
#include <QFileInfo>

namespace Sync {

namespace {

QString quoted(const QString &s)
{
    return QStringLiteral("\"") + s + QStringLiteral("\"");
}

QString nativePath(const QString &path)
{
    return QDir::toNativeSeparators(QDir::cleanPath(path));
}

const QString &badChars()
{
    static const QString s = QStringLiteral("\\/:*?\"<>|");
    return s;
}

} // namespace

bool isValidShareName(const QString &name)
{
    if (name.isEmpty() || name.size() > 80)
        return false;
    if (name.startsWith(QLatin1Char(' ')) || name.endsWith(QLatin1Char(' ')))
        return false;
    for (const QChar c : name) {
        if (c.unicode() < 0x20 || badChars().contains(c))
            return false;
    }
    return true;
}

QString suggestShareName(const QString &localPath)
{
    QString base = QFileInfo(nativePath(localPath)).fileName();
    if (base.isEmpty())
        base = QStringLiteral("SyncHub");
    QString out;
    for (const QChar c : base)
        out.append(c.unicode() < 0x20 || badChars().contains(c) ? QLatin1Char('_') : c);
    out = out.trimmed();
    if (out.isEmpty())
        out = QStringLiteral("SyncHub");
    return out.left(80);
}

QString shareScript(const QString &localPath, const QString &shareName, const QString &user)
{
    const QString path = nativePath(localPath);
    const QString nl   = QStringLiteral("\r\n");
    return QStringLiteral("@echo off") + nl
         + QStringLiteral("net share ") + shareName + QLatin1Char('=') + quoted(path)
         + QStringLiteral(" /grant:") + user + QStringLiteral(",change") + nl
         + QStringLiteral("if errorlevel 1 exit /b 1") + nl
         + QStringLiteral("icacls ") + quoted(path)
         + QStringLiteral(" /grant ") + quoted(user + QStringLiteral(":(OI)(CI)M")) + nl
         + QStringLiteral("exit /b %errorlevel%") + nl;
}

QStringList connectArgs(const QString &uncPath, const QString &user, const QString &password)
{
    QStringList args{QStringLiteral("use"), uncPath, QStringLiteral("/persistent:yes")};
    if (!user.isEmpty()) {
        args << QStringLiteral("/user:") + user;
        if (!password.isEmpty())
            args << password;
    }
    return args;
}

QString explainExitCode(int code, const QString &target)
{
    switch (code) {
    case 0:    return QStringLiteral("成功");
    case 2:    return QStringLiteral("已经存在或已经连接");
    case 5:    return QStringLiteral("拒绝访问：需要管理员权限（%1）").arg(target);
    case 53:   return QStringLiteral("找不到网络路径：确认主机名/IP 正确、在同一网络、防火墙放行 445");
    case 67:   return QStringLiteral("找不到共享名：主机上还没开这个共享，或名字拼写不同");
    case 86:   return QStringLiteral("密码不正确");
    case 1219: return QStringLiteral("同名连接已存在且用了别的账号，先执行 net use \"%1\" /delete").arg(target);
    case 1326: return QStringLiteral("登录失败：用户名或密码不对（要用主机上的账号，形如 主机名\\用户名）");
    default:   return QStringLiteral("命令失败（退出码 %1）").arg(code);
    }
}

} // namespace Sync
```

QString FolderTransport::probeError()
{
    if (m_sharedDir.isEmpty())
        return QStringLiteral("还没有填共享目录");

    if (!QDir(m_sharedDir).exists())
        return QStringLiteral("共享目录不存在或访问不到：%1（UNC 请先在资源管理器里打开它，确认能进并且没有再弹凭据框）")
            .arg(m_sharedDir);

    if (!QDir().mkpath(metaDir()))
        return QStringLiteral("共享目录能进，但建不了子目录 %1（只读共享或权限不足）").arg(metaDir());

    const QString probeFile = metaDir() + QStringLiteral("/.write-probe");
    QFile f(probeFile);
    if (!f.open(QIODevice::WriteOnly))
        return QStringLiteral("共享目录能进，但不可写：%1（共享权限和 NTFS 权限都要给「修改」；也可能是 Defender 受控文件夹访问拦住了）")
            .arg(probeFile);

    f.write("ok");
    f.close();
    QFile::remove(probeFile);
    return QString();
}

- [ ] **Step 2: 写实现**

```cpp
#include "core/os/ElevatedRun.h"

#include <string>

#include <windows.h>
#include <shellapi.h>

namespace Os {

int runElevated(const QString &commandLine, QString *error)
{
    if (commandLine.trimmed().isEmpty()) {
        if (error)
            *error = QStringLiteral("命令为空");
        return -1;
    }

    // cmd /c "<命令>"：命令里可能含引号，整体再包一层
    const std::wstring verb = L"runas";
    const std::wstring file = L"cmd.exe";
    std::wstring params = QStringLiteral("/c \"" + commandLine + QStringLiteral("\"")).toStdWString();

    SHELLEXECUTEINFOW sei{};
    sei.cbSize = sizeof(sei);
    sei.fMask  = SEE_MASK_NOCLOSEPROCESS | SEE_MASK_FLAG_NO_UI;
    sei.lpVerb = verb.c_str();
    sei.lpFile = file.c_str();
    sei.lpParameters = params.c_str();
    sei.nShow  = SW_HIDE;

    if (!ShellExecuteExW(&sei)) {
        const DWORD err = GetLastError();
        if (err == ERROR_CANCELLED) {
            if (error)
                *error = QStringLiteral("已取消管理员授权，共享没有开启");
            return -1000;
        }
        if (error)
            *error = QStringLiteral("无法启动提权进程（错误码 %1）").arg(static_cast<qulonglong>(err));
        return -1;
    }

    if (!sei.hProcess) {
        if (error)
            *error = QStringLiteral("提权进程没有句柄");
        return -1;
    }

    WaitForSingleObject(sei.hProcess, INFINITE);
    DWORD code = 0;
    GetExitCodeProcess(sei.hProcess, &code);
    CloseHandle(sei.hProcess);
    return static_cast<int>(code);
}

} // namespace Os
```

```cpp
QString SyncBridge::shareStatus() const { return m_shareStatus; }
bool    SyncBridge::shareBusy() const   { return m_shareBusy; }

void SyncBridge::setShareStatus(const QString &text)
{
    if (m_shareStatus == text)
        return;
    m_shareStatus = text;
    emit shareStatusChanged();
}

void SyncBridge::setShareBusy(bool b)
{
    if (m_shareBusy == b)
        return;
    m_shareBusy = b;
    emit shareBusyChanged();
}

QString SyncBridge::shareNameSuggestion() const
{
    return Sync::suggestShareName(toLocalPath(m_sharedDir));
}

void SyncBridge::shareFolder()
{
    if (m_shareBusy)
        return;

    const QString dir = toLocalPath(m_sharedDir).trimmed();
    if (dir.isEmpty()) {
        setShareStatus(QStringLiteral("先选一个本地文件夹作为共享目录"));
        emit shareOpFinished(false, m_shareStatus);
        return;
    }
    if (dir.startsWith(QStringLiteral("\\\\")) || dir.startsWith(QStringLiteral("//"))) {
        setShareStatus(QStringLiteral("这一侧是连接方：填对方的 \\\\主机\\共享名，不要在这里开共享"));
        emit shareOpFinished(false, m_shareStatus);
        return;
    }
    if (!QDir(dir).exists()) {
        setShareStatus(QStringLiteral("共享目录不存在：%1").arg(dir));
        emit shareOpFinished(false, m_shareStatus);
        return;
    }

    const QString name = Sync::suggestShareName(dir);
    const QString user = qEnvironmentVariable("USERNAME");
    const QString script = Sync::shareScript(dir, name, user);
    const QString batPath = QDir::temp().absoluteFilePath(
        QStringLiteral("assistant_share_%1.bat").arg(QDateTime::currentMSecsSinceEpoch()));

    {
        QFile f(batPath);
        if (!f.open(QIODevice::WriteOnly)) {
            setShareStatus(QStringLiteral("写不了临时脚本：%1").arg(batPath));
            emit shareOpFinished(false, m_shareStatus);
            return;
        }
        // cmd 按系统 OEM 代码页读 .bat，中文路径必须用本地 8 位编码写，不能写 UTF-8
        f.write(script.toLocal8Bit());
    }

    setShareBusy(true);
    setShareStatus(QStringLiteral("正在请求管理员权限…"));

    QThread *worker = QThread::create([this, batPath, dir, name] {
        QString err;
        const int code = Os::runElevated(QStringLiteral("\"%1\"").arg(QDir::toNativeSeparators(batPath)), &err);
        QFile::remove(batPath);

        QString msg;
        bool ok = false;
        if (code == 0) {
            ok  = true;
            msg = QStringLiteral("共享已开启：\\\\%1\\%2（把这一串发给对方，对方填进「对方共享目录」并点连接）")
                      .arg(localIpv4(), name);
        } else if (code == -1000 || code == -1) {
            msg = err;
        } else {
            msg = Sync::explainExitCode(code, dir);
        }

        QMetaObject::invokeMethod(this, [this, ok, msg] {
            setShareBusy(false);
            setShareStatus(msg);
            if (ok) {
                Logger::get().info("SyncBridge: 共享目录已开启");
                emit shareOpFinished(true, msg);
                sharedCheckResultRebuilt();
            } else {
                Logger::get().warn("SyncBridge: 开启共享失败：{}", msg.toStdString());
                emit shareOpFinished(false, msg);
            }
        }, Qt::QueuedConnection);
    });
    QObject::connect(worker, &QThread::finished, worker, &QObject::deleteLater);
    worker->start();
}
```

> `sharedCheckResultRebuilt()` 替换成已有的刷新动作：直接 `emit pathsChanged(); refreshChannel();`，再让 QML 自己调 `checkShared()`（QML 在 `shareOpFinished(true)` 里调）。删除对该不存在函数的调用。

- [ ] **Step 3: 实现连接**

```cpp
void SyncBridge::connectShared(const QString &user, const QString &password)
{
    if (m_shareBusy)
        return;

    const QString unc = m_sharedDir.trimmed();
    if (unc.isEmpty()) {
        setShareStatus(QStringLiteral("先填对方的 \\\\主机\\共享名"));
        emit shareOpFinished(false, m_shareStatus);
        return;
    }
    if (!unc.startsWith(QStringLiteral("\\\\")) && !unc.startsWith(QStringLiteral("//"))) {
        setShareStatus(QStringLiteral("连接方要填网络地址，形如 \\\\主机名\\共享名"));
        emit shareOpFinished(false, m_shareStatus);
        return;
    }

    setShareBusy(true);
    setShareStatus(QStringLiteral("正在连接 %1…").arg(unc));

    QProcess *p = new QProcess(this);
    p->setProgram(QStringLiteral("net"));
    p->setArguments(Sync::connectArgs(QDir::toNativeSeparators(unc), user, password));
    p->setProcessChannelMode(QProcess::MergedChannels);

    connect(p, &QProcess::finished, this, [this, p, unc](int code, QProcess::ExitStatus) {
        const QByteArray raw = p->readAll();
        p->deleteLater();

        QString msg;
        bool ok = false;
        if (code == 0 || code == 2) {
            ok  = true;
            msg = QStringLiteral("已连接 %1").arg(unc);
        } else {
            msg = Sync::explainExitCode(code, unc);
        }
        setShareBusy(false);
        setShareStatus(msg);

        if (ok) {
            setSharedDir(unc);
            startWatchers();
            refreshChannel();
            Logger::get().info("SyncBridge: net use 成功 {}", unc.toStdString());
        } else {
            Logger::get().warn("SyncBridge: net use 失败 code={} out={}", code, raw.toStdString());
        }
        emit shareOpFinished(ok, msg);
    });
    connect(p, &QProcess::errorOccurred, this, [this, p, unc](QProcess::ProcessError) {
        p->deleteLater();
        setShareBusy(false);
        setShareStatus(QStringLiteral("启动 net 命令失败：%1").arg(unc));
        emit shareOpFinished(false, m_shareStatus);
    });
    p->start();
}
```
