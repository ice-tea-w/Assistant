# 直发与待发队列（outbox）实施计划

**目标**：队友不在线时把"要发的东西"排队，对方上线后自动送达；同时恢复"发单个文件给队友"这个能力（旧投递区绑在共享目录上，随共享 UI 一起隐藏了）。

**架构**：待发队列持久化在 `sync.db` 的 `outbox` 表；发送走已有的 `NetTransport::push`，投递路径固定为 `_inbox/<发送设备>/<时间戳>_<原名>`；服务端把 `_inbox` 首段映射到第一个同步根；接收方把 `_inbox/<自己>/…` 领取到本地收件箱目录并从对端删除。

**技术栈**：C++20 / Qt6（QSqlDatabase、QSslSocket 已完成部分）/ QML。

---

## 设计决定（先看这五条）

1. **队列存 `sync.db`**，不新建 DB 文件（遵循「每功能一个 DB」的约定在这里不适用：队列与基线同生命周期，且要能一起备份）。
2. **投递路径固定 `_inbox/…`**，与同步目录解耦：发送方不关心对方的同步根叫什么、别名是什么。
3. **服务端对 `_inbox` 首段做特例**：`splitRel()` 见到首段是 `_inbox`（大小写不敏感）时，直接落到 `m_roots.first().localPath/_inbox/…`，不需要别名。
4. **引擎两侧都跳过含 `_inbox` / `_投递` 段的路径**，否则投递进来的文件会被当成同步内容再传一遍。
5. **不做投递回执**：`push` 成功即视为送达（文件已落在对方磁盘的 `_inbox` 下）。真正"领取"是接收方自己的动作。这一条是明确的简化，写进文档避免以后误以为是端到端确认。

**非目标**：断点续传、多接收人、投递权限（谁能给谁发）、投递回执与已读状态。

```cpp
bool SyncDatabase::enqueueOutbox(OutboxItem *item)
{
    if (!m_db.isOpen() || !item)
        return false;
    QSqlQuery q(m_db);
    q.prepare(QStringLiteral(
        "INSERT OR REPLACE INTO outbox (local_path, rel_path, size, created_at, attempts,"
        " next_attempt_at, last_error) VALUES (?, ?, ?, ?, ?, ?, ?)"));
    q.addBindValue(item->localPath);
    q.addBindValue(item->relPath);
    q.addBindValue(item->size);
    q.addBindValue(item->createdAt);
    q.addBindValue(item->attempts);
    q.addBindValue(item->nextAttemptAt);
    q.addBindValue(item->lastError);
    if (!q.exec())
        return false;
    item->id = q.lastInsertId().toLongLong();
    return true;
}

QList<OutboxItem> SyncDatabase::dueOutbox(qint64 now) const
{
    QList<OutboxItem> out;
    if (!m_db.isOpen())
        return out;
    QSqlQuery q(m_db);
    q.prepare(QStringLiteral(
        "SELECT id, local_path, rel_path, size, created_at, attempts, next_attempt_at,"
        " last_error FROM outbox WHERE next_attempt_at <= ? ORDER BY created_at"));
    q.addBindValue(now);
    if (!q.exec())
        return out;
    while (q.next()) {
        OutboxItem it;
        it.id            = q.value(0).toLongLong();
        it.localPath     = q.value(1).toString();
        it.relPath       = q.value(2).toString();
        it.size          = q.value(3).toLongLong();
        it.createdAt     = q.value(4).toLongLong();
        it.attempts      = q.value(5).toInt();
        it.nextAttemptAt = q.value(6).toLongLong();
        it.lastError     = q.value(7).toString();
        out.append(it);
    }
    return out;
}

QList<OutboxItem> SyncDatabase::outboxItems() const
{
    QList<OutboxItem> out;
    if (!m_db.isOpen())
        return out;
    QSqlQuery q(m_db);
    if (!q.exec(QStringLiteral(
            "SELECT id, local_path, rel_path, size, created_at, attempts, next_attempt_at,"
            " last_error FROM outbox ORDER BY created_at")))
        return out;
    while (q.next()) {
        OutboxItem it;
        it.id            = q.value(0).toLongLong();
        it.localPath     = q.value(1).toString();
        it.relPath       = q.value(2).toString();
        it.size          = q.value(3).toLongLong();
        it.createdAt     = q.value(4).toLongLong();
        it.attempts      = q.value(5).toInt();
        it.nextAttemptAt = q.value(6).toLongLong();
        it.lastError     = q.value(7).toString();
        out.append(it);
    }
    return out;
}

bool SyncDatabase::markOutboxSent(qint64 id)
{
    return removeOutbox(id);
}

bool SyncDatabase::markOutboxFailed(qint64 id, const QString &error, qint64 nextAttemptAt)
{
    if (!m_db.isOpen())
        return false;
    QSqlQuery q(m_db);
    q.prepare(QStringLiteral(
        "UPDATE outbox SET attempts = attempts + 1, last_error = ?, next_attempt_at = ?"
        " WHERE id = ?"));
    q.addBindValue(error);
    q.addBindValue(nextAttemptAt);
    q.addBindValue(id);
    return q.exec();
}

bool SyncDatabase::removeOutbox(qint64 id)
{
    if (!m_db.isOpen())
        return false;
    QSqlQuery q(m_db);
    q.prepare(QStringLiteral("DELETE FROM outbox WHERE id = ?"));
    q.addBindValue(id);
    return q.exec();
}

int SyncDatabase::outboxCount() const
{
    if (!m_db.isOpen())
        return 0;
    QSqlQuery q(m_db);
    if (!q.exec(QStringLiteral("SELECT COUNT(*) FROM outbox")) || !q.next())
        return 0;
    return q.value(0).toInt();
}
```

```cpp
bool isReservedDeliveryPath(const QString &relPath)
{
    const QStringList parts = relPath.split(QLatin1Char('/'), Qt::SkipEmptyParts);
    for (const QString &part : parts) {
        if (part.compare(QStringLiteral("_inbox"), Qt::CaseInsensitive) == 0

    }
    return false;
}
```

```cpp
QVariantList SyncBridge::outboxList() const { return m_outboxList; }

void SyncBridge::refreshOutbox()
{
    m_outboxList.clear();
    const QList<OutboxItem> items = m_db.outboxItems();
    for (const OutboxItem &it : items) {
        QVariantMap row;
        row.insert(QStringLiteral("id"), it.id);
        row.insert(QStringLiteral("name"), QFileInfo(it.localPath).fileName());
        row.insert(QStringLiteral("attempts"), it.attempts);
        row.insert(QStringLiteral("error"), it.lastError);
        row.insert(QStringLiteral("size"), it.size);
        m_outboxList.append(row);
    }
    emit outboxChanged();
}

void SyncBridge::queueSend(const QStringList &paths)
{
    if (!hasPeerCredentials()) {
        setNetStatus(QStringLiteral("先跟主机配对，才知道这些东西发给谁"));
        return;
    }
    const QDateTime stamp = QDateTime::currentDateTime();
    const QString sender = Sync::sanitizeDeviceName(m_deviceName);
    int queued = 0;
    for (const QString &raw : paths) {
        const QFileInfo fi(raw);
        if (!fi.isFile() || !fi.isReadable())
            continue;
        OutboxItem item;
        item.localPath = fi.absoluteFilePath();
        item.relPath = QStringLiteral("_inbox/%1/%2")
                           .arg(sender, Sync::makeDeliveryFileName(fi.fileName(), stamp));
        item.size      = fi.size();
        item.createdAt = QDateTime::currentSecsSinceEpoch();
        if (m_db.enqueueOutbox(&item))
            ++queued;
    }
    refreshOutbox();
    if (queued > 0) {
        setNetStatus(QStringLiteral("已排队 %1 个文件，正在尝试发送…").arg(queued));
        flushOutbox();
    }
}

void SyncBridge::removeOutboxItem(qint64 id)
{
    if (m_db.removeOutbox(id)) {
        refreshOutbox();
        setNetStatus(QStringLiteral("已从待发队列移除"));
    }
}
```

```cpp
Sync::Net::NetPeer SyncBridge::peerForSend() const
{
    Sync::Net::NetPeer peer;
    if (!m_config)
        return peer;
    const Sync::Net::PeerAddress parsed =
        Sync::Net::parsePeerAddress(m_config->syncPeerAddress(), static_cast<quint16>(m_config->syncPort()));
    if (!parsed.ok)
        return peer;
    peer.host               = parsed.host;
    peer.port               = parsed.port;
    peer.expectedFingerprint = m_config->syncPeerFingerprint();
    peer.deviceId           = m_config->syncDeviceId();
    peer.deviceSecret       = QByteArray::fromBase64(m_config->syncPeerSecret().toLatin1());
    peer.timeoutMs          = 8000;
    return peer;
}

void SyncBridge::flushOutbox()
{
    if (m_outboxBusy)
        return;
    const QList<OutboxItem> due = m_db.dueOutbox(QDateTime::currentSecsSinceEpoch());
    if (due.isEmpty()) {
        refreshOutbox();
        return;
    }
    const Sync::Net::NetPeer peer = peerForSend();
    if (peer.host.isEmpty() || peer.deviceSecret.isEmpty()) {
        setNetStatus(QStringLiteral("待发 %1 个文件：还没有可用的主机地址 / 口令").arg(due.size()));
        return;
    }

    m_outboxBusy = true;
    auto *worker = QThread::create([this, due, peer] {
        Sync::Net::NetTransport transport(peer);
        int sent = 0;
        int failed = 0;
        QString lastError;
        bool usable = transport.probe();
        for (const OutboxItem &item : due) {
            if (!usable) {
                ++failed;
                lastError = transport.lastError();
                const qint64 backoff = qMin<qint64>(60 * (item.attempts + 1), 600);
                m_db.markOutboxFailed(item.id, lastError,
                                      QDateTime::currentSecsSinceEpoch() + backoff);
                continue;
            }
            if (transport.push(item.localPath, item.relPath)) {
                m_db.markOutboxSent(item.id);
                ++sent;
            } else {
                ++failed;
                lastError = transport.lastError();
                const qint64 backoff = qMin<qint64>(60 * (item.attempts + 1), 600);
                m_db.markOutboxFailed(item.id, lastError,
                                      QDateTime::currentSecsSinceEpoch() + backoff);
            }
        }
        QMetaObject::invokeMethod(this, [this, sent, failed, lastError] {
            m_outboxBusy = false;
            refreshOutbox();
            if (sent > 0 && failed == 0)
                setNetStatus(QStringLiteral("已送达 %1 个文件").arg(sent));
            else if (sent > 0)
                setNetStatus(QStringLiteral("送达 %1 个，%2 个留在队列（%3）")
                                 .arg(sent).arg(failed).arg(lastError));
            else
                setNetStatus(QStringLiteral("主机关机或没监听，%1 个文件留在队列稍后重试").arg(failed));
            emit outboxAttemptFinished(sent, failed);
        }, Qt::QueuedConnection);
    });
    connect(worker, &QThread::finished, this, [worker] { worker->deleteLater(); });
    worker->start();
}
```

```cpp
int SyncBridge::fetchIncoming()
{
    const Sync::Net::NetPeer peer = peerForSend();
    if (peer.host.isEmpty() || peer.deviceSecret.isEmpty())
        return 0;
    const QString me = Sync::sanitizeDeviceName(m_deviceName);
    const QString prefix = QStringLiteral("_inbox/%1/").arg(me);
    const QString inbox = inboxPath();
    if (inbox.isEmpty())
        return 0;

    Sync::Net::NetTransport transport(peer);
    const QList<Sync::RemoteFileInfo> all = transport.listRemote();
    if (all.isEmpty())
        return 0;

    QStringList got;
    for (const Sync::RemoteFileInfo &f : all) {
        if (!f.relPath.startsWith(prefix))
            continue;
        if (m_db.isReceived(f.relPath)) {
            transport.removeRemote(f.relPath);
            continue;
        }
        const QString tmp = QDir(QDir::tempPath()).filePath(QStringLiteral("assistant-incoming.bin"));
        QFile::remove(tmp);
        if (!transport.fetch(f.relPath, tmp))
            continue;
        const QString placed = Sync::placeIncoming(tmp, inbox);
        QFile::remove(tmp);
        if (placed.isEmpty())
            continue;
        m_db.markReceived(f.relPath, f.hash);
        transport.removeRemote(f.relPath);
        got << QFileInfo(placed).fileName();
    }
    if (!got.isEmpty())
        emit filesReceived(got, inbox);
    return got.size();
}
```
